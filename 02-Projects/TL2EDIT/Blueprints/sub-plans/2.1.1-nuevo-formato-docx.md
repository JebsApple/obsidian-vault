## Objetivo
Cambiar el formato de exportación DOCX para que:
1. **Cada bloque = su propio párrafo** (eliminar lógica de `groupId` que unía bloques)
2. **Línea en blanco entre bloque y bloque** (`spacing: { after: 200 }`)
3. **Sin separación entre páginas** (todo corrido en 1 sección)
4. **Imágenes se mantienen** con `ImageRun`
5. **Apodos**: aplicar `applyNicknames()` al texto

## Formato final esperado

```
-diálogo

()pensamiento

**grito

[imagen: sticker.png]    ← imagen insertada como ImageRun

-diálogo siguiente
```

### 1. Re-escribir `buildPageParagraphs()` → eliminar, reemplazar por `buildBlockParagraphs()`

```typescript
import { applyNicknames } from '../hooks/useNicknames'
import type { NicknameRule } from '../types'

// NUEVA: procesa todos los bloques de todas las páginas sin separación
async function buildBlockParagraphs(
  pages: ComicPage[],
  config: BlockTypesConfig,
  nicknames?: NicknameRule[]
): Promise<Paragraph[]> {
  const paragraphs: Paragraph[] = []
  let firstBlock = true

  for (const page of pages.filter(p => p.status === 'success')) {
    const blocks = page.textBlocks.filter(b => b.exportEnabled)
    for (const block of blocks) {
      // Agregar línea en blanco entre bloques (excepto antes del primero)
      if (!firstBlock) {
        paragraphs.push(new Paragraph({ children: [new TextRun('')], spacing: { after: 60 } }))
      }
      firstBlock = false

      if (block.blockType === 'imagen') {
        if (block.attachedImage) {
          try {
            const { buffer, type } = dataUrlToUint8Array(block.attachedImage)
            paragraphs.push(new Paragraph({
              children: [new ImageRun({
                type: type as any,
                data: buffer,
                transformation: { width: 300, height: 200 }
              })],
              spacing: { after: 100 },
            }))
          } catch { /* skip broken image */ }
        }
        continue
      }

      const prefix = blockPrefix(block, config)
      const text = block.approvedText || block.originalText || ' '
      const finalText = nicknames ? applyNicknames(text, nicknames, 'translated') : text

      paragraphs.push(new Paragraph({
        children: [new TextRun(`${prefix}${finalText}`)],
        spacing: { after: 200 },
      }))
    }
  }

  return paragraphs
}
```

### 2. Re-escribir `buildDocxBlobMulti()`

```typescript
export async function buildDocxBlobMulti(
  pages: ComicPage[],
  config: BlockTypesConfig,
  nicknames?: NicknameRule[]
): Promise<Blob> {
  const legend = buildLegendParagraphs(config)
  const content = await buildBlockParagraphs(pages, config, nicknames)

  if (content.length === 0) {
    content.push(new Paragraph({ children: [new TextRun('')] }))
  }

  const doc = new Document({
    sections: [{ children: [...legend, ...content] }],
  })

  return Packer.toBlob(doc)
}
```
