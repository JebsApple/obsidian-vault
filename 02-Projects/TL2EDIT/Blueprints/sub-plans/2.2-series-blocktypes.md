# Sub-Plan 2.2: Series con Tipos de Bloque Propios

**Agente**: Sub-agente específico
**Dificultad**: MEDIA-ALTA | **Archivos**: 5 (1 nuevo hook, 1 nueva UI, 3 modificar) | **Estimación**: 6–8h

## Objetivo
Cada serie (manga/webtoon) puede tener sus propios overrides de prefijos, labels y fuentes para los tipos de bloque. Cuando una serie está activa, sus overrides reemplazan a los globales.

## Arquitectura

```
useSeries()  ← store a nivel módulo (localStorage)
  ├── seriesList: Series[]
  ├── currentSeries: Series | null
  ├── setCurrentSeries(id)
  ├── addSeries(name), removeSeries(id), updateSeries(id, partial)
  └── getEffectiveBlockType(blockTypeId, typeConfig) → {prefix, fontFamily, label}
```

Resolución: `resolveBlockType()` acepta `seriesOverrides` opcional → serie > global > default.

### 2. Modificar `resolveBlockType()` en `src/types.ts`

```typescript
export function resolveBlockType(
  typeConfig: BlockTypesConfig,
  blockType: string,
  seriesOverrides?: SeriesBlockTypeOverride[]
): { label: string; prefix: string; fontFamily: string; icon: string } {
  // 1. Resolver el tipo base (global)
  const base = resolveBaseBlockType(typeConfig, blockType)  // extraer lógica actual a esta función

  // 2. Aplicar override de serie si existe
  if (seriesOverrides) {
    const override = seriesOverrides.find(o => o.blockTypeId === blockType)
    if (override) {
      return {
        ...base,
        label: override.label ?? base.label,
        prefix: override.prefix ?? base.prefix,
        fontFamily: override.fontFamily ?? base.fontFamily,
      }
    }
  }

  return base
}

// Lógica original extraída
function resolveBaseBlockType(typeConfig: BlockTypesConfig, blockType: string): { label: string; prefix: string; fontFamily: string; icon: string } {
  const builtin = BLOCK_TYPES.find(t => t.id === blockType)
  if (builtin) {
    const settings = typeConfig.types[builtin.id]
    return {
      label: builtin.id === 'custom' ? typeConfig.customLabel : builtin.label,
      prefix: settings?.prefix ?? builtin.defaultPrefix,
      fontFamily: settings?.fontFamily ?? '',
      icon: builtin.icon,
    }
  }
  const custom = typeConfig.customTypes.find(t => t.id === blockType)
  if (custom) {
    return { label: custom.label, prefix: custom.prefix, fontFamily: custom.fontFamily, icon: 'ph-tag' }
  }
  const fallback = BLOCK_TYPES[0]
  return { label: fallback.label, prefix: fallback.defaultPrefix, fontFamily: '', icon: fallback.icon }
}
```

### 4. `src/components/SeriesPanel.tsx` — CREAR NUEVO

```tsx
import { useState } from 'react'
import { useSeries } from '../hooks/useSeries'
import { BLOCK_TYPES } from '../types'

export default function SeriesPanel() {
  const { seriesList, currentSeries, setCurrentSeries, addSeries, removeSeries, updateSeries, addOverride, removeOverride, getActiveOverrides } = useSeries()
  const [newName, setNewName] = useState('')

  return (
    <div className="panel" style={{ padding: 'var(--space-3)', display: 'flex', flexDirection: 'column', gap: 8 }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
        <i className="ph-duotone ph-books" style={{ fontSize: 16 }} />
        <h6 style={{ margin: 0, flex: 1 }}>Series</h6>
      </div>

      {/* Selector de serie activa */}
      <select className="input" value={currentSeries?.id ?? ''}
        onChange={e => setCurrentSeries(e.target.value || null)} style={{ fontSize: 13 }}>
        <option value="">Sin serie</option>
        {seriesList.map(s => <option key={s.id} value={s.id}>{s.name}</option>)}
      </select>

      {/* Agregar serie nueva */}
      <div style={{ display: 'flex', gap: 4 }}>
        <input className="input" placeholder="Nombre de serie" value={newName}
          onChange={e => setNewName(e.target.value)} style={{ flex: 1, fontSize: 13 }} />
        <button className="btn btn-primary" disabled={!newName.trim()}
          onClick={() => { addSeries(newName.trim()); setNewName('') }} style={{ fontSize: 13 }}>
          Agregar
        </button>
      </div>

      {/* Overrides de la serie activa */}
      {currentSeries && (
        <div style={{ display: 'flex', flexDirection: 'column', gap: 4, fontSize: 12 }}>
          <span className="text-muted">Overrides de prefijo para "{currentSeries.name}"</span>
          {BLOCK_TYPES.filter(t => t.id !== 'imagen').map(def => {
            const override = currentSeries.blockTypeOverrides.find(o => o.blockTypeId === def.id)
            return (
              <div key={def.id} style={{ display: 'flex', gap: 4, alignItems: 'center' }}>
                <span style={{ width: 80 }}>{def.label}</span>
                <input className="input" placeholder={def.defaultPrefix}
                  value={override?.prefix ?? ''}
                  onChange={e => {
                    if (e.target.value) addOverride(currentSeries.id, { blockTypeId: def.id, prefix: e.target.value })
                    else removeOverride(currentSeries.id, def.id)
                  }}
                  style={{ width: 60, fontSize: 11 }} />
                <span className="text-muted">default: {def.defaultPrefix}</span>
              </div>
            )
          })}
        </div>
      )}
    </div>
  )
}
```
