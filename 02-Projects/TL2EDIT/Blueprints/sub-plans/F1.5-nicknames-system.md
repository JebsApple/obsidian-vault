# Sub-Plan F1.5: Sistema de Apodos

**Agente**: Sub-agente D
**Dificultad**: MEDIA | **Archivos**: 3 (1 nuevo hook, 1 nueva UI, 1 modificar types) | **Estimación**: 3–4h

## Objetivo
Crear sistema de apodos que reemplacen nombres de personajes tanto en el texto original como en el traducido. Un apodo tiene: nickname (cómo se muestra), originalPattern (patrón en japonés/chino), translatedPattern (patrón en español).

### 2. `src/hooks/useNicknames.ts` — CREAR NUEVO

Mismo patrón que `src/hooks/useReplacements.ts` (store a nivel módulo + `useSyncExternalStore`):

```typescript
import { useCallback, useSyncExternalStore } from 'react'
import type { NicknameRule } from '../types'

const STORAGE_KEY = 'tl2edit-nicknames'

function loadNicknames(): NicknameRule[] {
  try {
    const raw = localStorage.getItem(STORAGE_KEY)
    if (!raw) return []
    const parsed = JSON.parse(raw)
    if (!Array.isArray(parsed)) return []
    return parsed.map(sanitizeRule).filter(Boolean) as NicknameRule[]
  } catch { return [] }
}

function sanitizeRule(raw: unknown): NicknameRule | null {
  if (!raw || typeof raw !== 'object') return null
  const r = raw as Partial<NicknameRule>
  const id = typeof r.id === 'string' && r.id ? r.id : `nn-${Math.random().toString(36).substring(2, 9)}`
  return {
    id,
    nickname: typeof r.nickname === 'string' ? r.nickname.slice(0, 100) : '',
    originalPattern: typeof r.originalPattern === 'string' ? r.originalPattern.slice(0, 200) : '',
    translatedPattern: typeof r.translatedPattern === 'string' ? r.translatedPattern.slice(0, 200) : '',
    enabled: typeof r.enabled === 'boolean' ? r.enabled : true,
  }
}

function saveNicknames(rules: NicknameRule[]): void {
  try { localStorage.setItem(STORAGE_KEY, JSON.stringify(rules)) } catch {}
}

// Store a nivel módulo
let current: NicknameRule[] = loadNicknames()
const listeners = new Set<() => void>()

function notify(): void {
  saveNicknames(current)
  listeners.forEach(l => l())
}

function subscribe(listener: () => void): () => void {
  listeners.add(listener)
  return () => listeners.delete(listener)
}

function getSnapshot(): NicknameRule[] { return current }

export interface UseNicknamesResult {
  nicknames: NicknameRule[]
  addRule: () => void
  removeRule: (id: string) => void
  updateRule: (id: string, partial: Partial<Pick<NicknameRule, 'nickname' | 'originalPattern' | 'translatedPattern'>>) => void
  toggleRule: (id: string) => void
}

export function useNicknames(): UseNicknamesResult {
  const nicknames = useSyncExternalStore(subscribe, getSnapshot)

  const addRule = useCallback(() => {
    current = [...current, {
      id: `nn-${Math.random().toString(36).substring(2, 9)}`,
      nickname: '', originalPattern: '', translatedPattern: '', enabled: true,
    }]
    notify()
  }, [])

  const removeRule = useCallback((id: string) => {
    current = current.filter(r => r.id !== id)
    notify()
  }, [])

  const updateRule = useCallback((id: string, partial: Partial<Pick<NicknameRule, 'nickname' | 'originalPattern' | 'translatedPattern'>>) => {
    current = current.map(r => r.id === id ? { ...r, ...partial } : r)
    notify()
  }, [])

  const toggleRule = useCallback((id: string) => {
    current = current.map(r => r.id === id ? { ...r, enabled: !r.enabled } : r)
    notify()
  }, [])

  return { nicknames, addRule, removeRule, updateRule, toggleRule }
}

/** Función utilitaria para aplicar apodos a un texto. Se usa desde exportDocx. */
export function applyNicknames(text: string, nicknames: NicknameRule[], mode: 'original' | 'translated'): string {
  let result = text
  for (const n of nicknames) {
    if (!n.enabled) continue
    const pattern = mode === 'original' ? n.originalPattern : n.translatedPattern
    if (!pattern || !n.nickname) continue
    // Reemplazar solo palabras completas (word boundary)
    result = result.replace(new RegExp(escapeRegex(pattern), 'g'), n.nickname)
  }
  return result
}

function escapeRegex(s: string): string {
  return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
}
```

### 3. `src/components/NicknamesPanel.tsx` — CREAR NUEVO

Panel similar a `ReplacementsPanel.tsx` con 3 campos por fila:

```tsx
import { useState } from 'react'
import { useNicknames } from '../hooks/useNicknames'

export default function NicknamesPanel() {
  const { nicknames, addRule, removeRule, updateRule, toggleRule } = useNicknames()
  const [importText, setImportText] = useState('')

  return (
    <div className="panel" style={{ padding: 'var(--space-3)', display: 'flex', flexDirection: 'column', gap: 8 }}>
      <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
        <i className="ph-duotone ph-tag" style={{ fontSize: 16 }} />
        <h6 style={{ margin: 0, flex: 1 }}>Apodos</h6>
        <button className="btn btn-ghost btn-icon" onClick={addRule}>
          <i className="ph-duotone ph-plus" style={{ fontSize: 14 }} />
        </button>
      </div>

      {nicknames.length === 0 && (
        <p className="text-muted" style={{ fontSize: 12, margin: 0 }}>
          Agrega apodos para personajes. Se reemplazarán tanto en el texto original como en el traducido.
        </p>
      )}

      <div style={{ display: 'flex', flexDirection: 'column', gap: 4, maxHeight: 300, overflowY: 'auto' }}>
        {nicknames.map(n => (
          <div key={n.id} style={{ display: 'flex', gap: 4, alignItems: 'center', fontSize: 12 }}>
            <input
              type="checkbox"
              checked={n.enabled}
              onChange={() => toggleRule(n.id)}
              style={{ flex: 'none' }}
            />
            <input
              className="input"
              placeholder="Apodo"
              value={n.nickname}
              onChange={e => updateRule(n.id, { nickname: e.target.value })}
              style={{ width: 80, fontSize: 11 }}
            />
            <input
              className="input"
              placeholder="Nombre original"
              value={n.originalPattern}
              onChange={e => updateRule(n.id, { originalPattern: e.target.value })}
              style={{ flex: 1, fontSize: 11 }}
            />
            <input
              className="input"
              placeholder="Nombre traducido"
              value={n.translatedPattern}
              onChange={e => updateRule(n.id, { translatedPattern: e.target.value })}
              style={{ flex: 1, fontSize: 11 }}
            />
            <button className="btn btn-ghost btn-icon" onClick={() => removeRule(n.id)}>
              <i className="ph-duotone ph-trash" style={{ fontSize: 13 }} />
            </button>
          </div>
        ))}
      </div>
    </div>
  )
}
```
