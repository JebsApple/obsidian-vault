### 1. Exponential Backoff en `driveFetch()`

Envolver `driveFetch()` actual con reintentos:

```typescript
const RETRYABLE_STATUSES = [429, 500, 502, 503]
const MAX_RETRIES = 3
const BASE_DELAY_MS = 1000

async function driveFetchWithRetry(
  accessToken: string,
  url: string,
  init?: RequestInit,
  signal?: AbortSignal,
  retries = MAX_RETRIES
): Promise<Response> {
  for (let attempt = 0; attempt <= retries; attempt++) {
    try {
      const res = await driveFetch(accessToken, url, init, signal)
      // Si es exitoso o no es retryable, devolver
      if (res.ok || !RETRYABLE_STATUSES.includes(res.status)) {
        return res
      }
      // Si es el último intento, devolver el error
      if (attempt === retries) {
        return res
      }
      // Esperar con backoff exponencial
      const delay = BASE_DELAY_MS * Math.pow(2, attempt)
      await new Promise(r => setTimeout(r, delay))
    } catch (err) {
      if (attempt === retries) throw err
      // Si es error de red, esperar y reintentar
      if (err instanceof TypeError) { // network error
        const delay = BASE_DELAY_MS * Math.pow(2, attempt)
        await new Promise(r => setTimeout(r, delay))
        continue
      }
      throw err
    }
  }
  throw new Error('Max retries exceeded')
}
```

**IMPORTANTE**: No reemplazar `driveFetch()` directamente. Crear la nueva función y usarla donde aplique. Mantener `driveFetch()` para operaciones que no deben reintentarse (POST de creación).
